\chapter{The Algebraic Part}\label{ch:algebraic-part}

The Hermite reduction writes $f = Dg + h$, where $h$ has a squarefree
denominator with respect to an integral basis. For rational functions that is
nearly the end: $h$ is a polynomial plus a proper fraction with squarefree
denominator, the polynomial is integrated termwise, and the fraction
contributes only logarithms. For algebraic functions the analogue of the
polynomial part is harder to see, because an integral element of $E$ can have
poles at infinity of an order that its coordinates do not show.

Trager's solution~\cite[\S4.3]{trager1984} is to arrange that the integrand
has no poles at infinity to begin with, by a change of variable, and to use an
integral basis that is \emph{normal at infinity}. Then one of two things
happens. Either the Hermite reduction has introduced a pole at infinity, in
which case the integral is not elementary; or it has not, and then $g$ is
the whole algebraic part of the integral, and what remains is a sum of
logarithms or nothing.

The setting is that of Chapter~\ref{ch:liouville}: $E = K(x)[y]/(F)$ with $F$
absolutely irreducible, $D = d/dx$, and places are places of $E_{\Kbar}$. We
write $\OO$ for the integral closure of $K[x]$ in $E$.

\section{The algebraic part is integral}\label{sec:algpart-integral}

\begin{lemma}\label{lem:simple-poles}\standing{proved}{}
Let $w$ be an integral basis of $E$ over $K(x)$, and let $h \in E$ have a
squarefree denominator with respect to $w$. Then $\ord{P}(h) \ge -r_P$ at
every finite place $P$; equivalently, $h\,dx$ has at most simple poles at the
finite places. Conversely, if $h \in E$ and $h\,dx$ has at most simple poles
at the finite places, then the denominator of $h$ with respect to $w$ is
squarefree.
\end{lemma}

\begin{proof}
Let $d$ be the denominator and $P$ a finite place over $x_0$. Since $d$ is
squarefree, $\ord{P}(d) \le \ord{P}(x - x_0) = r_P$, and $d h \in \OO$, so
$\ord{P}(h) \ge -r_P$. The equivalence is~\eqref{eq:order-dx}. Conversely,
let $d^*$ be the squarefree part of the denominator $d$ of $h$. If
$\ord{P}(h) \ge -r_P$ at every finite $P$, then $\ord{P}(d^* h) \ge 0$ at
every finite $P$, so $d^* h \in \OO = \OO_w$, and $d$ divides $d^*$.
\end{proof}

\begin{proposition}\label{prop:v-integral}\standing{proved}{}
Let $w$ be an integral basis of $E$ over $K(x)$, and let $h \in E$ have a
squarefree denominator with respect to $w$. If
\[
  h = Dv + \sum_{i=1}^k c_i\, \frac{Du_i}{u_i}
\]
with $v \in E$, $c_i \in \Kbar$ and $u_i \in E_{\Kbar}^*$, then $v \in \OO$.
\end{proposition}

\begin{proof}
Let $P$ be a finite place. By Lemma~\ref{lem:simple-poles}, $\ord{P}(h) \ge
-r_P$. By~\ref{D:log} and~\eqref{eq:order-dx}, $\ord{P}(Du_i / u_i) \ge -r_P$
for each $i$. Hence $\ord{P}(Dv) \ge -r_P$. If $\ord{P}(v)$ were negative,
Lemma~\ref{lem:order-derivative}(i) would give $\ord{P}(Dv) = \ord{P}(v) - r_P
< -r_P$. So $v$ has no pole at any finite place, and $v \in \OO$.
\end{proof}

So after the Hermite reduction the algebraic part of the integral of $h$, if
there is one, is $v = \sum_i p_i w_i$ with polynomials $p_i$. It ``corresponds
to the polynomial part of the integral of rational functions'', in the words
of~\cite[\S2.4]{bronstein1998tutorial}. Finding the $p_i$ directly would need
bounds on their degrees. The rest of the chapter shows how to make them
constants.

\section{Integrality at infinity}\label{sec:algpart-infinity}

For $a \in K(x)^*$ let $\deg(a)$ be the degree of its numerator minus that of
its denominator. The rational functions that have no pole at infinity form
the ring
\[
  K(x)_\infty = \{ a \in K(x)^* : \deg(a) \le 0 \} \cup \{0\} ,
\]
a local ring whose maximal ideal is generated by $1/x$. An element of $E$ is
\emph{integral at infinity}\index{integral at infinity} if it has no pole at any infinite place; these
elements form the ring $\OO_\infty$, which is the integral closure of
$K(x)_\infty$ in $E$ and a free $K(x)_\infty$-module of rank $n$
(Theorem~\ref{thm:places}(iii)).

\begin{definition}\label{def:normal-infinity}
A basis $w = (w_1, \dots, w_n)$ of $E$ over $K(x)$ is \emph{normal at
infinity}\index{normal at infinity} if there are $r_1, \dots, r_n \in K(x)^*$ such that $(r_1 w_1,
\dots, r_n w_n)$ is a basis of $\OO_\infty$ over $K(x)_\infty$.
\end{definition}

This is Trager's definition~\cite[\S2.3]{trager1984}. Since every nonzero
rational function is a power of $x$ times a unit of $K(x)_\infty$, the $r_i$
can be taken to be powers of $x$: there are integers $\tau_1, \dots, \tau_n$,
determined by $w$, such that
\begin{equation}\label{eq:tau}
  (x^{-\tau_1} w_1, \dots, x^{-\tau_n} w_n)
  \text{ is a basis of } \OO_\infty \text{ over } K(x)_\infty .
\end{equation}
\seealg{NormalizeAtInfinity} (Section~\ref{sec:ib-infinity}) turns any
integral basis into one that is also normal at infinity, and finds the
$\tau_i$. What makes
such a basis useful is that poles at infinity cannot cancel between its
terms.

\begin{lemma}\label{lem:termwise}\standing{proved}{}
Let $w$ be a basis of $E$ over $K(x)$ that is normal at infinity, with
$\tau_i$ as in~\eqref{eq:tau}, and let $f_1, \dots, f_n \in K(x)$. Then
\begin{align*}
  \sum_{i=1}^n f_i w_i \in \OO_\infty
  &\iff f_i w_i \in \OO_\infty \text{ for each } i \\
  &\iff f_i = 0 \text{ or } \deg(f_i) \le -\tau_i \text{ for each } i .
\end{align*}
If moreover $w_i \in \OO$ for each $i$, then $\tau_i \ge 0$, with equality
exactly when $w_i \in K$.
\end{lemma}

\begin{proof}
Write $\sum_i f_i w_i = \sum_i (f_i x^{\tau_i}) (x^{-\tau_i} w_i)$. Since the
$x^{-\tau_i} w_i$ form a basis of $\OO_\infty$ over $K(x)_\infty$, the sum is
in $\OO_\infty$ if and only if each $f_i x^{\tau_i}$ is in $K(x)_\infty$,
that is, $\deg(f_i) \le -\tau_i$; and the same criterion applied to the
single term $f_i w_i$ shows that it is then in $\OO_\infty$.

Suppose $w_i \in \OO$. If $\tau_i < 0$, then $w_i = x^{\tau_i} (x^{-\tau_i}
w_i)$ would vanish at every infinite place and have no pole anywhere, so
$w_i = 0$ by~\ref{D:const}. If $\tau_i = 0$, then $w_i$ has no pole at all, so
$w_i \in \Kbar \cap E = K$; and if $w_i \in K^*$ then $w_i$ is a unit of
$\OO_\infty$, so $\tau_i = 0$.
\end{proof}

\section{Trager's theorem on poles at infinity}\label{sec:algpart-trager}

\begin{theorem}[Trager]\label{thm:algebraic-part}\standing{proved}{}
Let $w$ be an integral basis of $E$ over $K(x)$ that is normal at infinity.
Let $f \in E$ be such that $f\,dx$ has no pole at any infinite place, and let
$(g, h) = \seealg{HermiteReduce}(f, w)$. Suppose that $f$ has an elementary
integral over $E$. Then
\begin{enumerate}[label=\textup{(\roman*)}]
\item $g$ is integral at infinity, and $h\,dx$ has no pole at any infinite
  place;
\item there are $c_1, \dots, c_k \in \Kbar$ and $u_1, \dots, u_k \in
  E_{\Kbar}^*$ such that $h = \sum_i c_i\, Du_i / u_i$;
\item if all the residues of $f\,dx$ are zero, then $h = 0$.
\end{enumerate}
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:liouville}, $f = Dv_0 + \sum_i c_i\, Du_i/u_i$ with $v_0
\in E$, $c_i \in \Kbar$ and $u_i \in E_{\Kbar}^*$, and by
Corollary~\ref{cor:second-kind} the sum can be taken empty when all the
residues of $f\,dx$ are zero. Let $v = v_0 - g$, so that
\[
  h = Dv + \sum_i c_i\, \frac{Du_i}{u_i} .
\]
The denominator of $h$ with respect to $w$ is squarefree, so $v \in \OO$ by
Proposition~\ref{prop:v-integral}. Since $w$ is an integral basis, $v = \sum_i
p_i w_i$ with $p_i \in K[x]$. On the other hand $g = \sum_i g_i w_i$, where
each $g_i \in K(x)$ is a sum of fractions $b/v^{m-1}$ with $\deg(b) <
\deg(v)$, produced by the steps of \seealg{HermiteReduce}; so $g_i = 0$ or
$\deg(g_i) < 0$. Therefore
\[
  v_0 = \sum_{i=1}^n (p_i + g_i)\, w_i,
  \qquad \deg(p_i + g_i) = \deg(p_i) \ge 0 \text{ whenever } p_i \ne 0 .
\]
Now consider an infinite place $P$. The differential $dv_0 = f\,dx - \sum_i
c_i\, du_i/u_i$ has order at least $-1$ there, because $f\,dx$ has no pole
at $P$ and by~\ref{D:log}. By~\ref{D:exact}, a pole of $v_0$ at $P$ would be
a pole of $dv_0$ of order at least $2$. So $v_0 \in \OO_\infty$, and by
Lemma~\ref{lem:termwise}, for each $i$ with $p_i + g_i \ne 0$,
\[
  \deg(p_i + g_i) \le -\tau_i \le 0 .
\]
If $p_i \ne 0$ this forces $\deg(p_i) = 0$ and $\tau_i = 0$, that is, $p_i
\in K$ and $w_i \in K$. So $v$ is a sum of products of constants: $v \in
K$, and $Dv = 0$. This proves~(ii), and~(iii) when the sum is empty.

For~(i), $g = v_0 - v$ is integral at infinity. By~\ref{D:exact}, $dg$ then
has no pole at an infinite place, and neither does $h\,dx = f\,dx - dg$.
\end{proof}

This is Theorem~4.5 and Corollary~4.6 of~\cite{trager1984}. Read backwards,
part~(i) is a test for non-integrability, and part~(iii) is a decision
procedure for a problem of independent interest.

\begin{corollary}[integration in $E$]\label{cor:in-field}\standing{proved}{}
With $w$, $f$, $g$ and $h$ as in Theorem~\ref{thm:algebraic-part}, there is a
$v_0 \in E$ with $f = Dv_0$ if and only if $h = 0$, and then $v_0 = g$ up to
an additive constant.
\end{corollary}

\begin{proof}
If $f = Dv_0$ with $v_0 \in E$, then all the residues of $f\,dx$ are zero
by~\ref{D:exact}, and $f$ has an elementary integral, so $h = 0$ by
Theorem~\ref{thm:algebraic-part}(iii). Conversely, if $h = 0$ then $f = Dg$.
\end{proof}

To apply the theorem we must be able to tell whether $h\,dx$ has a pole at
infinity. By~\eqref{eq:order-dx} it has none exactly when
\begin{equation}\label{eq:no-pole-infinity}
  \ord{P}(h) \ge r_P + 1 \quad \text{at every infinite place } P .
\end{equation}

\begin{lemma}\label{lem:pole-test}\standing{proved}{}
Let $w$ be normal at infinity with $\tau_i$ as in~\eqref{eq:tau}, and $h =
\sum_i h_i w_i$ with $h_i \in K(x)$. If no infinite place of $E$ is
ramified, then $h\,dx$ has no pole at infinity if and only if
\[
  h_i = 0 \quad\text{or}\quad \deg(h_i) \le -\tau_i - 2
  \qquad (1 \le i \le n).
\]
In general, the condition $\deg(h_i) \le -\tau_i - 2$ for all $i$ is
sufficient, and the condition $\deg(h_i) \le -\tau_i - 1$ for all $i$ is
necessary.
\end{lemma}

\begin{proof}
At an infinite place $\ord{P}(x) = -r_P$. So $x^2 h \in \OO_\infty$ means
$\ord{P}(h) \ge 2 r_P$ for all infinite $P$, which implies
\eqref{eq:no-pole-infinity} and is equivalent to it when every $r_P = 1$;
and~\eqref{eq:no-pole-infinity} implies $\ord{P}(h) \ge r_P$, that is, $x h
\in \OO_\infty$. Lemma~\ref{lem:termwise} translates $x^2 h \in \OO_\infty$
and $x h \in \OO_\infty$ into the two conditions on degrees.
\end{proof}

\section{The change of variable}\label{sec:algpart-change}

The hypothesis of Theorem~\ref{thm:algebraic-part}, that $f\,dx$ has no pole
at infinity, is arranged by moving to infinity a point where $f\,dx$ has no
pole.

\begin{lemma}\label{lem:change-variable}\standing{proved}{}
Let $b$ be a basis of $E$ over $K(x)$ consisting of integral elements, let $f
\in E$, and let $N \in K$ not be a root of the denominator of $f$ with respect
to $b$. Put $z = 1/(x - N)$ and $\tilde f = -f/z^2$. Then $E$ is a finite
extension of $K(z) = K(x)$, and with respect to the variable $z$:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $f\,dx = \tilde f\,dz$, and this differential has no pole at any place
  that is infinite for $z$;
\item $f$ has an elementary integral over $(E, d/dx)$ if and only if $\tilde
  f$ has one over $(E, d/dz)$, and $\int f\,dx = \int \tilde f\,dz$;
\item if $N$ is not a root of the discriminant of $F$ with respect to $y$,
  and $F$ is monic in $y$, then no place that is infinite for $z$ is
  ramified over $K(z)$.
\end{enumerate}
\end{lemma}

\begin{proof}
We have $dx = -dz/z^2$, which gives the first equality, and $d/dz = -z^{-2}\,
d/dx$ on $E$. The places that are infinite for $z$ are the places over $x =
N$. At such a place $f$ has no pole, because the denominator of $f$ does not
vanish at $N$ and the elements of $b$ are integral over $K[x]$; and $dx$ has
no pole at a finite place, by~\eqref{eq:order-dx}. This proves~(i). For~(ii),
$f = Dv + \sum_i c_i\, Du_i / u_i$ holds for $d/dx$ if and only if the same
equation multiplied by $-z^{-2}$ holds for $d/dz$, and the constants of the
two derivations are the same. For~(iii), the places over $x = N$ are
unramified when $F(N, y)$ has $n$ distinct roots
(Theorem~\ref{thm:places}(iv)).
\end{proof}

An $N$ as in the lemma always exists, since only finitely many elements of
$K$ are excluded and $K$ is infinite. The first part of the integration
algorithm is then as follows; it is the first five of the six steps
of~\cite[\S2.4]{bronstein1998tutorial}.

\begin{algo}{AlgebraicPart}{f}{the algebraic part of an integral}
\Spec{Given $f \in E$, return either ``not elementary'', when $\int f\,dx$ is
not elementary over $E$; or a variable $z$ with $K(z) = K(x)$, an integral
basis $w$ of $E$ over $K(z)$ normal at infinity, and $g, h \in E$ such that
$\int f\,dx = g + \int h\,dz$, the denominator of $h$ with respect to $w$ is
squarefree, $h\,dz$ has no pole at infinity, and $\int f\,dx$ is elementary
if and only if $h$ is a linear combination with constant coefficients of
logarithmic derivatives $(du/dz)/u$.}
\State $b \gets$ a basis of $E$ over $K(x)$ made of integral elements
\State $N \gets$ an element of $K$ that is a root neither of the denominator of $f$ with respect to $b$ nor of the discriminant of $F$
\State $z \gets 1/(x - N)$, \ $\tilde f \gets -f / z^2$ \Comment{Lemma~\ref{lem:change-variable}}
\State $w \gets$ an integral basis of $E$ over $K(z)$, normal at infinity \Comment{Chapter~\ref{ch:integral-bases}}
\State $(g, h) \gets \alg{HermiteReduce}(\tilde f, w)$ \Comment{with $D = d/dz$}
\State $(h_1, \dots, h_n) \gets$ the coordinates of $h$ in the basis $w$
\If{$h_i \ne 0$ and $\deg(h_i) > -\tau_i - 2$ for some $i$} \Comment{Lemma~\ref{lem:pole-test}}
  \State \Return ``not elementary''
\EndIf
\State \Return $(z, w, g, h)$
\end{algo}

The algorithm is correct by Lemma~\ref{lem:change-variable},
Theorem~\ref{thm:algebraic-part} and Lemma~\ref{lem:pole-test}. When it
returns $h = 0$ the integral is $g$; and by Corollary~\ref{cor:in-field},
$\int f\,dx$ is in $E$ if and only if that happens.

Three remarks. First, the change of variable is a device for meeting the
hypothesis of Theorem~\ref{thm:algebraic-part}; when $f\,dx$ already has no
pole at infinity, the theorem applies with $z = x$, and
Lemma~\ref{lem:pole-test} still gives a test when infinity is unramified.
Second, the condition on the discriminant only serves to make that test
exact; without it, one tests~\eqref{eq:no-pole-infinity} place by place.
Third, the lazy Hermite reduction can be used in place of the fourth and
fifth lines, starting from any basis of integral elements: if it returns $h =
0$, the integral has been found without an integral basis, and otherwise an
integral basis normal at infinity is computed at that point and the reduction
is run again with respect to it~\cite[\S2.4]{bronstein1998tutorial}.

\begin{example}\label{ex:running-algpart}
We apply \seealg{AlgebraicPart} to $f = x/\sqrt{1 - x^3}$, in the field $E =
\Q(x)[y]/(y^2 - (1 - x^3))$. By Example~\ref{ex:running-differentials},
$f\,dx$ has a double pole at infinity, so the change of variable is needed.
\begin{enumerate}
\item With $b = (1, y)$, the denominator of $f = -xy/(x^3 - 1)$ is $x^3 - 1$,
  and the discriminant of $y^2 - (1 - x^3)$ is $4 (1 - x^3)$. We may take $N =
  0$.
\item $z = 1/x$. Put $Y = z^2 y$, so that $Y^2 = z^4 - z$ and $E =
  \Q(z)[Y]/(Y^2 - (z^4 - z))$. Then
  \[
    \tilde f = -\frac{f}{z^2} = -\frac{1}{z\,Y} = \frac{-Y}{z^2 (z^3 - 1)} .
  \]
\item Since $z^4 - z$ is squarefree, $w = (1, Y)$ is an integral basis, and it
  is normal at infinity with $(\tau_1, \tau_2) = (0, 2)$
  (Chapter~\ref{ch:radical}). There are two infinite places, unramified, at
  which $Y/z^2 = \pm 1$.
\item \seealg{HermiteReduce}: $d = z^2 (z^3 - 1)$, so $v = z$, $m = 2$ and $u
  = z^3 - 1$. From $2 Y\, DY = 4z^3 - 1$ we get $S_2 = u\,(z\, DY - Y) =
  (z^3 + \frac12)\, Y$. The solution of $f_2 S_2 = -Y$ is $f_2 = -2/(2 z^3 +
  1)$, so $q = 2z^3 + 1$, $\rho = 1$, $b_2 = -2$ and
  \[
    g = -\frac{2Y}{z}, \qquad
    h = \tilde f - Dg = \frac{2 z\, Y}{z^3 - 1} .
  \]
\item $h_2 = 2z/(z^3 - 1)$ has degree $-2$, and $-\tau_2 - 2 = -4$. So $h\,dz$
  has a pole at infinity, and the algorithm returns ``not elementary''.
\end{enumerate}
Hence
\[
  \int \frac{x\,dx}{\sqrt{1 - x^3}}
\]
is not an elementary function. In the original variable the reduction reads
\[
  \int \frac{x\,dx}{\sqrt{1 - x^3}}
  = -\frac{2 \sqrt{1 - x^3}}{x} - 2 \int \frac{dx}{x^2 \sqrt{1 - x^3}} ,
\]
which is correct and looks like a step backwards: the pole of the integrand
at infinity has been exchanged for a double pole at $x = 0$. That exchange is
the point. The original differential was regular at $x = 0$, so an
elementary integral could not have had an algebraic part with a pole there,
and Theorem~\ref{thm:algebraic-part} says that the reduction would then not
have produced one.
\end{example}

\begin{example}\label{ex:running-algpart-2}
For $f = 1/(x \sqrt{1 - x^3})$ and for $f = 1/(1 - x^3)^{3/2}$ no change of
variable is needed, because $f\,dx$ has no pole at infinity
(Example~\ref{ex:running-differentials}; for the second, the order at
$\infty$ is $9 - 3 = 6$). The basis $w = (1, y)$ is an integral basis normal
at infinity, with $(\tau_1, \tau_2) = (0, 2)$: the one infinite place is
ramified, $x^{-2} y$ has order $1$ there and $x^{-1} y$ has order $-1$.

For $f = 1/(x\sqrt{1 - x^3})$, the Hermite reduction returns $g = 0$ and $h =
f$, and $h\,dx$ has order $2$ at $\infty$. Theorem~\ref{thm:algebraic-part}
says that the integral, if elementary, is a sum of logarithms with no
algebraic part. We find them in Chapter~\ref{ch:logarithmic-part}.

For $f = 1/(1 - x^3)^{3/2}$, Example~\ref{ex:running-hermite} gave $g =
2x/(3y)$ and $h = 1/(3y)$. Here $g$ is integral at infinity, as part~(i) of
the theorem requires of an elementary integral, and $h\,dx$ has no pole at
infinity; so this test does not decide. But all residues of $f\,dx$ are
zero and $h \ne 0$, so by part~(iii) the integral is not elementary, in
agreement with Example~\ref{ex:first-kind}.
\end{example}

\section*{Exercises}
\addcontentsline{toc}{section}{Exercises}

\begin{exercise}
Let $E = K(x)$, so that $n = 1$ and $w = (1)$. What do
Theorem~\ref{thm:algebraic-part} and \seealg{AlgebraicPart} say about the
integral of a rational function? Why is the change of variable not used in
\VolI{Chapter~2}?
\end{exercise}

\begin{exercise}
Run \seealg{AlgebraicPart} on $f = 1/\sqrt{1 - x^2}$ with $N = 0$, and
describe the differential $h\,dz$ that it returns.
\end{exercise}

\begin{exercise}
Show that for an infinite place $P$ and $u \in E_{\Kbar}^*$ with $\ord{P}(u)
\ne 0$, $\ord{P}(Du) = \ord{P}(u) + r_P$. This is Lemma~4.3
of~\cite{trager1984}, the counterpart at infinity of
Lemma~\ref{lem:order-derivative}.
\end{exercise}

\begin{exercise}
Use Corollary~\ref{cor:in-field} to decide whether $\int (3x^3 - 2)\,dx /
(x^2 \sqrt{1 - x^3})$ is an algebraic function.
\end{exercise}
