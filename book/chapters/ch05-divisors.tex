\chapter{Divisors and Their Arithmetic}\label{ch:divisors}

The logarithmic part of an integral is found by asking, of a divisor
$\delta$ of degree $0$, whether $\delta$, or $2\delta$, or $3\delta$, \dots\
is the divisor of a function (Chapter~\ref{ch:logarithmic-part}). This
chapter makes the question computable: it represents divisors by ideals,
gives Trager's test for principality together with the function when there
is one, and describes the alternatives.

Throughout, $K$ is a field of characteristic $0$, $E = K(x)[y]/(F)$ with $F$
absolutely irreducible, $\OO$ is the integral closure of $K[x]$ in $E$, and
places and divisors are those of $E_{\Kbar}$, as in Part~II. A divisor is
\emph{finite} if its support consists of finite places.

\section{Divisors and ideals}\label{sec:div-ideals}

A \emph{fractional ideal} of $\OO$ is a nonzero $\OO$-submodule $I$ of $E$
such that $a I \subseteq \OO$ for some nonzero $a \in K[x]$. It is a free
$K[x]$-module of rank $n$. For a finite divisor $A = \sum_P n_P P$ let
\[
  I(A) = \{ b \in E : \ord{P}(b) \ge n_P \text{ for every finite place } P \} .
\]

\begin{theorem}\label{thm:divisor-ideal}\standing{cited}{\cite[Proposition~3.2.9]{stichtenoth2009}; \cite[Proposition~6.1]{trager1984}}
Let $K$ be algebraically closed. Then $A \mapsto I(A)$ is a bijection between
the finite divisors of $E$ and the fractional ideals of $\OO$, and $I(A + B)
= I(A)\, I(B)$, $I(-A) = I(A)^{-1}$. If $h_1, \dots, h_s \in E$ are such that
$\min_i \ord{P}(h_i) = n_P$ at every finite place $P$, then $I(A) = \OO h_1 +
\dots + \OO h_s$.
\end{theorem}

This is the statement that $\OO$ is a Dedekind domain whose nonzero prime
ideals are the finite places: every fractional ideal is a product of powers
of primes, and the exponents are the $n_P$. The last sentence is how
divisors enter the algorithm: by Proposition~\ref{prop:residue-ideal}, the
divisor $\Delta_\alpha$ of the places where the integrand has residue
$\alpha$ is $I(\Delta_\alpha) = \OO d + \OO (dh - \alpha d')$.

When $K$ is not algebraically closed, a finite divisor that is stable under
the automorphisms of $\Kbar$ over $K$ corresponds to a fractional ideal of
the integral closure of $K[x]$ in $E$, and the computations below are
carried out there. The residue divisors need the field $K'$ generated by the
residues and no more (Theorem~\ref{thm:liouville-residues}).

\paragraph{Computing with ideals.} A fractional ideal is stored as a
$K[x]$-basis, that is, an $n \times n$ matrix over $K(x)$ of coordinates with
respect to a fixed integral basis $w$. If $I = \sum_i \OO h_i$, a
$K[x]$-basis of $I$ is found by row reduction of the $sn \times n$ matrix of
coordinates of the products $w_j h_i$ (Section~\ref{sec:prelim-hermite}). The
product of two ideals is generated by the products of their generators, and
the inverse of an ideal is computed as described in
Section~\ref{sec:ib-criterion}. So $I(n\delta)$ can be computed for any
integer $n$ from the ideals $I(\Delta_\alpha)$.

Trager keeps the generators few~\cite[\S5.3]{trager1984}. He represents a
divisor $A$ whose support lies over the roots of a polynomial $a(x)$ by a
pair $(h, a)$, where $h \in E$ has $\ord{P}(h) = n_P$ at the places over the
roots of $a$; such an $h$ is obtained from $dh - \alpha d'$ by adding a
random integer multiple of $d$, to make its zeros simple. Pairs for $A$ and
$B$ with the same $a$ multiply to a pair for $A + B$, namely $(h_A h_B, a)$,
and divide similarly after the poles of the quotient away from the roots of
$a$ have been cleared.

\section{Testing for principality}\label{sec:div-principal}

\begin{theorem}[Trager]\label{thm:principal}\standing{proved}{}
Let $A$ be a finite divisor of degree $0$, and let $b = (b_1, \dots, b_n)$ be
a $K[x]$-basis of $I(A)$ that is normal at infinity, with $\tau_i$ as
in~\eqref{eq:tau}. Then:
\begin{enumerate}[label=\textup{(\roman*)}]
\item $A$ is principal if and only if $\tau_i \le 0$ for some $i$, that is, if
  and only if some $b_i$ has no pole at infinity.
\item If $b_i$ has no pole at infinity, then $A = (b_i)$.
\item At most one $\tau_i$ is $\le 0$, and it is then $0$.
\end{enumerate}
\end{theorem}

\begin{proof}
Let $u \in I(A)$ be nonzero and without poles at infinity. Then $\ord{P}(u)
\ge n_P$ at every finite place and $\ord{P}(u) \ge 0$ at every infinite one,
so $(u) \ge A$. Both divisors have degree $0$
(Theorem~\ref{thm:riemann-roch}(i)), so $(u) = A$. Conversely, if $A = (u)$
then $u \in I(A)$ and $u$ has neither zero nor pole at infinity. So $A$ is
principal if and only if $I(A) \cap \OO_\infty \ne \{0\}$, and any nonzero
element of the intersection is a generator.

Now let $u = \sum_i p_i b_i$ with $p_i \in K[x]$ be nonzero. By
Lemma~\ref{lem:termwise}, $u \in \OO_\infty$ if and only if $\deg(p_i) \le
-\tau_i$ for each $i$ with $p_i \ne 0$. Such $p_i$ exist if and only if some
$\tau_i \le 0$, and then $p_i = 1$ will do. This proves~(i) and~(ii). For~(iii),
the intersection is the space $\LL(-A)$ taken over $K$, whose dimension is at
most $1$ by Theorem~\ref{thm:riemann-roch}(iv), and by the above its
dimension is $\sum_{\tau_i \le 0} (1 - \tau_i)$.
\end{proof}

This is Theorem~6.3 of~\cite{trager1984}.

\begin{algo}{PrincipalGenerator}{I}{test a divisor of degree $0$ for principality}
\Spec{Given a $K[x]$-basis of the fractional ideal $I = I(A)$ of a finite
divisor $A$ of degree $0$, return $u \in E$ with $A = (u)$, or ``not
principal''.}
\State $v \gets$ a local integral basis at infinity \Comment{Section~\ref{sec:ib-infinity}}
\State $(b, \tau) \gets \alg{NormalizeAtInfinity}(I, v)$
\If{$\tau_i \le 0$ for some $i$} \Return $b_i$ \EndIf
\State \Return ``not principal''
\end{algo}

The search of Chapter~\ref{ch:logarithmic-part} is then a loop.

\begin{algo}{PrincipalMultiple}{I, B}{search for a principal multiple}
\Spec{Given the ideal $I = I(\delta)$ of a finite divisor $\delta$ of degree
$0$ and an integer $B \ge 1$, return $(n, u)$ with $1 \le n \le B$ least such
that $n \delta$ is principal, and $n \delta = (u)$; or ``none'' if no such $n$
exists.}
\State $J \gets I$
\For{$n \gets 1$ \textbf{to} $B$}
  \State $u \gets \alg{PrincipalGenerator}(J)$
  \If{$u$ is not ``not principal''} \Return $(n, u)$ \EndIf
  \State $J \gets J \cdot I$
\EndFor
\State \Return ``none''
\end{algo}

Chapter~\ref{ch:finite-order} supplies a $B$ such that ``none'' means that no
multiple of $\delta$ is principal, and a way to find the only possible $n$
without testing them all.

\begin{example}\label{ex:principal-running}
We run the search for the residue divisor of
Example~\ref{ex:running-log}: $E = \Q(x)[y]/(y^2 + x^3 - 1)$ and $\delta = P -
Q$ with $P = (0, 1)$, $Q = (0, -1)$. The integral basis is $(1, y)$; there is
one infinite place, where $x$ has order $-2$ and $y$ has order $-3$, and
$v = (1, y/x^2)$ is a local integral basis at infinity
(Theorem~\ref{thm:radical-basis}).

Since $(x) = P + Q - 2\infty$, we have $I(n\delta) = x^{-n}\, I(2n P)$. An
element $a + b\,y$ of $\OO$ vanishes to order $2n$ at $P$ when $a + b\,Y
\equiv 0 \pmod{x^{2n}}$, where $Y = 1 - \frac12 x^3 - \frac18 x^6 - \dots$ is
the expansion of $y$ at $P$; so $I(2nP)$ has the $K[x]$-basis $(x^{2n},\; y -
Y_n)$ with $Y_n$ the truncation of $Y$ modulo $x^{2n}$.

\emph{$n = 1$.} $Y_1 = 1$ and $I(\delta)$ has basis
\[
  b = \Bigl( x,\; \frac{y - 1}{x} \Bigr),
  \qquad M = \begin{pmatrix} x & 0 \\ -1/x & x \end{pmatrix}
\]
in terms of $v$. So $\kappa = (1, 1)$ and $N = \bigl(\begin{smallmatrix} 1 &
0 \\ 0 & 1 \end{smallmatrix}\bigr)$, which is invertible: $b$ is normal at
infinity with $\tau = (1, 1)$, and $\delta$ is not principal.

\emph{$n = 2$.} $Y_2 = 1 - \frac12 x^3$ and $I(2\delta)$ has basis
\[
  b = \Bigl( x^2,\; \frac{y - 1 + \frac12 x^3}{x^2} \Bigr),
  \qquad M = \begin{pmatrix} x^2 & 0 \\ \frac12 x - x^{-2} & 1 \end{pmatrix} .
\]
Now $\kappa = (2, 1)$ and $N = \bigl(\begin{smallmatrix} 1 & 0 \\ 1/2 & 0
\end{smallmatrix}\bigr)$ is singular, with $c = (1, -2)$. The index with the
greater $\kappa$ is $i_0 = 1$, and $b_1$ is replaced by
\[
  b_1 - 2x\, b_2 = -\frac{2\,(y - 1)}{x},
  \qquad\text{with row } \bigl( 2/x,\; -2x \bigr) .
\]
Then $\kappa = (1, 1)$ and $N = \bigl(\begin{smallmatrix} 0 & -2 \\ 1/2 & 0
\end{smallmatrix}\bigr)$ is invertible. So $\tau = (1, 1)$ and $2\delta$ is
not principal.

\emph{$n = 3$.} $Y_3 = 1 - \frac12 x^3$ again, and $I(3\delta)$ has basis
\[
  b = \Bigl( x^3,\; \frac{y - 1 + \frac12 x^3}{x^3} \Bigr),
  \qquad M = \begin{pmatrix} x^3 & 0 \\ \frac12 - x^{-3} & x^{-1} \end{pmatrix} .
\]
Here $\kappa = (3, 0)$ and $N = \bigl(\begin{smallmatrix} 1 & 0 \\ 1/2 & 0
\end{smallmatrix}\bigr)$ is singular, with $c = (1, -2)$ and $i_0 = 1$; $b_1$
is replaced by $b_1 - 2x^3 b_2 = -2\,(y - 1)$, with row $(2, -2x^2)$ and
$\kappa_1 = 2$. Now $N = \bigl(\begin{smallmatrix} 0 & -2 \\ 1/2 & 0
\end{smallmatrix}\bigr)$ is invertible and $\tau = (2, 0)$. Since $\tau_2 =
0$, the divisor $3\delta$ is principal, with generator
\[
  b_2 = \frac{2y - 2 + x^3}{2x^3} = \frac12 \cdot \frac{y - 1}{y + 1} ,
\]
the last equality because $(y - 1)(y + 1) = -x^3$. This is the function found
by inspection in Example~\ref{ex:running-log}, up to the constant factor,
which a logarithmic derivative does not see.
\end{example}

\section{Other methods}\label{sec:div-others}

The test of Theorem~\ref{thm:principal} is one way to compute a
Riemann--Roch space: $I(A) \cap \OO_\infty$ is $\LL(-A)$. There are others,
and the integration algorithm can use any of them.

\emph{Coates' algorithm.} Coates~\cite{coates1970} constructs the functions
with prescribed orders at given places from the Puiseux expansions at those
places, by linear algebra on the coefficients. Davenport's integration
algorithm~\cite{davenport1981} uses it for the principality test, where
Trager uses ideals. Volcheck~\cite{volcheck1994} adds in the Jacobian of a
plane curve with arbitrary singularities, extending the method of adjoints
of Brill and Noether with Puiseux expansions. We have read only the
introductions of these papers.

\emph{Hess's algorithm.\later} Hess~\cite{hess2002} computes Riemann--Roch
spaces ``without the Brill--Noether method of adjoints or any series
expansions'', and builds on that the reduction of divisors and computation in
the divisor class group. It is the modern replacement for the arithmetic of
this chapter, and for a new implementation it is the paper to read first. We
have read its abstract and introduction only, and have not reproduced it.

\section{Hyperelliptic curves}\label{sec:div-hyperelliptic}

For $y^2 = f(x)$ with $f$ squarefree of degree $2g + 1$, divisor classes have
a normal form in which the search for a principal multiple is cheap. We
follow Cantor~\cite{cantor1987}, who refers to Mumford for the proofs. Let
$K$ be algebraically closed. The curve has one place $\infty$ at infinity,
and for a point $P = (x_0, y_0)$ we write $P' = (x_0, -y_0)$; then $P + P' -
2\infty = (x - x_0)$ is principal.

A divisor $D = \sum_{i=1}^r P_i - r\,\infty$ with finite points $P_i = (x_i,
y_i)$ is \emph{semireduced} if no $P_i$ equals $P_j'$ for $j \ne i$, and
\emph{reduced} if moreover $r \le g$. A semireduced divisor is given by a
pair of polynomials $(a, b)$: $a = \prod_i (x - x_i)$, and $b$ is the
polynomial of degree less than $\deg(a)$ with $b \equiv y$ at each $P_i$,
with multiplicity; equivalently $a$ divides $b^2 - f$. We write $D =
\operatorname{div}(a, b)$. In the language of Section~\ref{sec:div-ideals},
$\sum_i P_i$ is the divisor of the ideal $\OO a + \OO (y - b)$.

\begin{theorem}\label{thm:mumford}\standing{cited}{\cite[\S2]{cantor1987}}
Every divisor of degree $0$ is equivalent, modulo principal divisors, to
exactly one reduced divisor $\operatorname{div}(a, b)$. In particular it is
principal if and only if its reduced form is $\operatorname{div}(1, 0)$.
\end{theorem}

\begin{algo}{CantorCompose}{a_1, b_1, a_2, b_2}{composition of semireduced divisors}
\Spec{Given semireduced divisors $\operatorname{div}(a_1, b_1)$ and
$\operatorname{div}(a_2, b_2)$ on $y^2 = f(x)$, return a semireduced
$\operatorname{div}(a, b)$ equivalent to their sum.}
\State $(d, h_1, h_2, h_3) \gets$ $d = \gcd(a_1, a_2, b_1 + b_2) = h_1 a_1 + h_2 a_2 + h_3 (b_1 + b_2)$
\State $a \gets a_1 a_2 / d^2$
\State $b \gets \bigl( h_1 a_1 b_2 + h_2 a_2 b_1 + h_3 (b_1 b_2 + f) \bigr) / d \bmod a$
\State \Return $(a, b)$
\end{algo}

\begin{algo}{CantorReduce}{a, b}{reduction of a semireduced divisor}
\Spec{Given a semireduced $\operatorname{div}(a, b)$ on $y^2 = f(x)$ of genus
$g$, return the reduced divisor equivalent to it.}
\While{$\deg(a) > g$}
  \State $a \gets (f - b^2) / a$
  \State $b \gets -b \bmod a$
\EndWhile
\State \Return $(a / \lc(a),\; b)$
\end{algo}

These are the formulas (C1)--(C3) and the classical reduction of
\cite[\S3 and~\S4]{cantor1987}, where their correctness is proved; the
reduction replaces $D$ by $-((b - y) - D)$, the divisor of the other zeros of
$y - b$, and each pass lowers $\deg(a)$ by at least $2$ while it exceeds $g +
1$. Cantor's paper also gives a faster reduction, which we omit.

So on a hyperelliptic curve $n\delta$ is computed by repeated composition
and reduction, in the manner of repeated squaring, and tested by comparing
with $(1, 0)$. Bertrand's integration algorithm~\cite{bertrand1995} rests on
this (Section~\ref{sec:radical-hyperelliptic}).

\begin{example}\label{ex:cantor-running}
The curve $y^2 = 1 - x^3$ has $g = 1$ and $f = 1 - x^3$. The divisor $\delta =
P - Q$ of Example~\ref{ex:principal-running} is equivalent to $2(P -
\infty)$, because $P + Q - 2\infty = (x)$. Now $P - \infty =
\operatorname{div}(x, 1)$.
\begin{enumerate}
\item Doubling: $a_1 = a_2 = x$, $b_1 = b_2 = 1$, so $d = \gcd(x, x, 2) = 1$
  with $h_3 = \frac12$. Then $a = x^2$ and $b = \frac12 (1 + f) \bmod x^2 = 1$.
  Reducing: $a \gets (f - 1)/x^2 = -x$, $b \gets -1$. So $2(P - \infty) \sim
  \operatorname{div}(x, -1) = Q - \infty$, which is not $(1, 0)$: $\delta$ is
  not principal.
\item $2\delta \sim 2 \operatorname{div}(x, -1)$, and by the same computation
  with the sign of $b$ changed this is $\operatorname{div}(x, 1) = P -
  \infty$: not principal.
\item $3\delta \sim \operatorname{div}(x, -1) + \operatorname{div}(x, 1)$.
  Here $b_1 + b_2 = 0$, so $d = \gcd(x, x, 0) = x$ and $a = x^2/x^2 = 1$: the
  sum is $\operatorname{div}(1, 0)$, and $3\delta$ is principal.
\end{enumerate}
This agrees with Example~\ref{ex:principal-running}. It decides principality
and does not by itself produce the function $u$; that comes from keeping
track of the functions $x - x_0$ and $y - b$ by which the divisor was changed
at each step.
\end{example}

\begin{exercise}
In Example~\ref{ex:principal-running}, compute $I(\delta)$ as the product
$I(P)\, I(Q)^{-1}$, with $I(P) = \OO x + \OO (y - 1)$ and $I(Q) = \OO x + \OO
(y + 1)$, using the procedures of Section~\ref{sec:ib-criterion}, and verify
that the result is the module found there.
\end{exercise}

\begin{exercise}
Use \seealg{PrincipalGenerator} to show that on $y^2 = x^2 + 1$ the divisor
$P_+ - P_-$ of Example~\ref{ex:log-sqrt2} is principal, and recover the
function $u_1$ found there.
\end{exercise}

\begin{exercise}
On $y^2 = x^8 + 1$, with $P = (0, 1)$ and $Q = (0, -1)$, the divisor $P - Q$
has two infinite places to contend with, so Cantor's representation does not
apply as it stands. Compute $I(n(P - Q))$ for $n = 1, \dots, 4$ as in
Example~\ref{ex:principal-running} and decide which are principal. This
verifies the statement taken from the tutorial in
Example~\ref{ex:radical-log}.
\end{exercise}
